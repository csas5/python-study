Git = 代码版本管理工具
GIT command
git add .					把修改的文件放入 Git 暂存区
git status					查看Git当前状态
git commit -m				创建一个版本记录  -m：给这次提交写一句说明
git remote add origin			给本地Git连接一个远程GitHub仓库  origin：远程仓库的别名=https://github.com/csas5/python-study.git
git branch -M main				修改分支名称    -M：Move/rename   强制修改名称。意为：把当前分支改成main
git config --global user.name "你的GitHub用户名" 设置Git作者名字。 config:修改Git设置。--两个横杠：表示：长参数。global全局。意思：整个电脑生效。
git config --global user.email “your email”	设置提交邮箱。
git config --list				查看所有配置。list:列出来。
git remote -v				查看远程仓库。-v:显示详细信息。
git push -u origin main			上传代码到GitHub。-u:upstream:建立跟踪关系    第一次：git push -u origin main   以后git push
git config --global http.proxy http://		设置HTTP代理。意思：Git访问网络：走：Clash代理。https.proxyHTTPS代理。GitHub使用：HTTPS协议。HTTP请求。
git config --global https.proxy http://		HTTPS请求。
git config --global --unset http.proxy		取消HTTP
